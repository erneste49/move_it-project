
2.	Query to get phone number and fullname of guest that came with cash 
 SELECT guest.guest_phone, guest.guest_fullname FROM guest JOIN gift
ON gift.guest_id = guest.guest_id
WHERE gift.gift_name = 'Cash';



3.	Query to get all guest from Zamfara state
SELECT * FROM guest JOIN state
ON guest.guest_state = state.state_id
WHERE state.state_name = 'Zamfara';


4. 	Query to get all details of female guest
SELECT * FROM guest
WHERE guest_gender = 'Female';

5. Query to get guests that didn't supply phone number
SELECT * FROM guest WHERE guest_phone = '';
